import datetime,hashlib,json,pathlib,re,shutil
out=pathlib.Path('/Users/tedczj/workspace/term4u/docs/redesign/phase1-acceptance/m5-m6/recovery/network-20260912')
raw=pathlib.Path('/tmp/term4u-net-final-02')
collect=json.loads((raw/'result.json').read_text())
packet=(raw/'packets.log').read_text();dns=(raw/'dns.log').read_text()
summary={'capture':collect,'raw_local_files':{},'runs':{},'controls':{}}
for f in raw.glob('*.log'):summary['raw_local_files'][str(f)]={'sha256':hashlib.sha256(f.read_bytes()).hexdigest(),'bytes':f.stat().st_size}
summary['packet_footer']=[x for x in packet.splitlines() if re.match(r'^\d+ packets ',x)]
summary['zero_kernel_drops']='0 packets dropped by kernel' in packet
for name in ['gui-final','tui-final']:
 r=json.loads((out/(name+'-run.json')).read_text());pids=set(r['pids'])
 start=datetime.datetime.fromisoformat(r['start']).astimezone();end=datetime.datetime.fromisoformat(r['end']).astimezone()
 def packet_in_window(line):
  m=re.match(r'(\d\d:\d\d:\d\d\.\d+)',line)
  if not m:return False
  t=datetime.datetime.fromisoformat(str(start.date())+'T'+m[1]).replace(tzinfo=start.tzinfo)
  return start<=t<=end
 def dns_in_window(line):
  try:t=datetime.datetime.fromisoformat(line[:23]).replace(tzinfo=start.tzinfo)
  except ValueError:return False
  return start<=t<=end
 pkts=[l for l in packet.splitlines() if packet_in_window(l) and any(int(x) in pids for x in re.findall(r'proc [^,:]*:(\d+)',l))]
 queries=[l for l in dns.splitlines() if dns_in_window(l) and any(int(x) in pids for x in re.findall(r'(?:client pid: |PID\[)(\d+)',l))]
 (out/(name+'-packets.log')).write_text('\n'.join(pkts)+ ('\n' if pkts else ''))
 (out/(name+'-dns.log')).write_text('\n'.join(queries)+ ('\n' if queries else ''))
 files=[];log_hits=[]
 for f in pathlib.Path(r['home']).rglob('*'):
  if f.is_file():
   files.append({'path':str(f.relative_to(r['home'])),'bytes':f.stat().st_size})
   if f.suffix=='.log':
    content=f.read_text(errors='replace');(out/(name+'-application.log')).write_text(content)
    log_hits.extend(l for l in content.splitlines() if re.search(r'retry|upload|telemetry|sentry|crash|denied|reject',l,re.I))
 (out/(name+'-files.json')).write_text(json.dumps(files,indent=2)+'\n')
 summary['runs'][name]={'duration_seconds':(end-start).total_seconds(),'known_pids':sorted(pids),'packet_lines':len(pkts),'dns_records':len(queries),'log_keyword_hits':log_hits,'complete_child_coverage':False,'status':'NO_ACTIVITY_OBSERVED_FOR_TRACKED_PIDS_ONLY'}
for name in ['final-calibration-probe','child-calibration','end-calibration']:
 r=json.loads((out/(name+'.json')).read_text());ids={r['pid']}
 if 'child' in r:ids.add(r['child']['pid'])
 pl=[l for l in packet.splitlines() if any(int(x) in ids for x in re.findall(r'proc [^,:]*:(\d+)',l))]
 dl=[l for l in dns.splitlines() if any(int(x) in ids for x in re.findall(r'(?:client pid: |PID\[)(\d+)',l))]
 (out/(name+'-packets.log')).write_text('\n'.join(pl)+'\n');(out/(name+'-dns.log')).write_text('\n'.join(dl)+'\n')
 summary['controls'][name]={'pids':sorted(ids),'packet_lines':len(pl),'dns_records':len(dl)}
summary['status']='INCOMPLETE: Endpoint Security child coverage unavailable; not full C8'
(out/'network-analysis.json').write_text(json.dumps(summary,indent=2)+'\n')
shutil.copyfile(raw/'result.json',out/'capture-result.json')
print(json.dumps({'runs':summary['runs'],'footer':summary['packet_footer'],'controls':summary['controls']},indent=2))
