import subprocess,time,pathlib,json,sys,datetime,signal,os
signal.signal(signal.SIGINT,signal.SIG_DFL)
signal.signal(signal.SIGTERM,signal.SIG_DFL)
try: os.kill(84360,signal.SIGKILL)
except ProcessLookupError: pass
run=pathlib.Path(sys.argv[1]);run.mkdir(exist_ok=True)
commands={'packets':['/usr/sbin/tcpdump','-i','pktap,all','-nn','-k','INPD','-l','tcp or udp'],'dns':['/usr/bin/log','stream','--level','debug','--style','compact','--predicate','process == "mDNSResponder"'],'exec':['/usr/bin/fs_usage','-w','-f','exec','-t','850']}
ps={};files={};start=datetime.datetime.now(datetime.timezone.utc).isoformat()
for name,cmd in commands.items():
 files[name]=(run/(name+'.log')).open('w');ps[name]=subprocess.Popen(cmd,stdout=files[name],stderr=subprocess.STDOUT)
time.sleep(2);(run/'ready').write_text(start)
deadline=time.monotonic()+840
while time.monotonic()<deadline and not pathlib.Path(str(run)+'.stop').exists():time.sleep(1)
result={}

 except subprocess.TimeoutExpired:p.kill();r=p.wait()
 files[name].close();result[name]={'command':commands[name],'exit_code':r}
(run/'result.json').write_text(json.dumps({'start':start,'end':datetime.datetime.now(datetime.timezone.utc).isoformat(),'collectors':result},indent=2)+'\n')
